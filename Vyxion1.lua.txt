    TrackedCharacters = {},

function Prediction:GetPredictedPosition(part)
    local data = self.TrackedCharacters[part.Parent]
    if not data then return part.Position end
    local ping = Client:GetNetworkPing()
    local pingClamped = MathClamp(ping, 0, 0.35)
    local travelTime = pingClamped + 0.05
    if self.ModelSettings.SimplifiedNetworkTimeModel then
        travelTime = travelTime * 0.85
    end
    local vel = data.Velocity or Vector3.zero
    local accel = data.Acceleration or Vector3.zero
    local predicted = part.Position + (vel * travelTime)
    if self.ModelSettings.AddAccelerationComponent then
        predicted = predicted + (accel * (travelTime * travelTime * 0.5))
    end
    return predicted
end
function Prediction:RegisterCharacter(char)
    if not char or self.TrackedCharacters[char] then return end
    local data = {
        LastPosition = char:GetPivot().Position,
        LastTime = tick(),
        Velocity = Vector3.zero,
        Acceleration = Vector3.zero,
        LastVelocity = Vector3.zero,
    }
    self.TrackedCharacters[char] = data
    local conn
    conn = RunService.Heartbeat:Connect(function(dt)
        if not char or not char.Parent then
            conn:Disconnect()
            self.TrackedCharacters[char] = nil
            return
        end
        local currentPos = char:GetPivot().Position
        local currentTime = tick()
        local deltaTime = MathMax(currentTime - data.LastTime, 0.016)
        if deltaTime > 0 then
            local newVel = (currentPos - data.LastPosition) / deltaTime
            if self.ModelSettings.UseCustomVelocityResolver then
                data.Velocity = data.Velocity and (data.Velocity * 0.7 + newVel * 0.3) or newVel
            else
                data.Velocity = newVel
            end
            if data.Velocity and data.LastVelocity then
                data.Acceleration = (data.Velocity - data.LastVelocity) / deltaTime
                if data.Acceleration.Magnitude > 100 then
                    data.Acceleration = data.Acceleration.Unit * 100
                end
            end
            data.LastVelocity = data.Velocity
            data.LastPosition = currentPos
        end
        data.LastTime = currentTime
    end)
end
local function GetDistanceRoot(target)
	local RootPos = RootPart.Position
	local TargetType = typeof(target)
	if TargetType == "Vector3" then
		return (target - RootPos).Magnitude
	elseif TargetType == "CFrame" then
		return (target.Position - RootPos).Magnitude
	elseif TargetType ~= "Instance" then
		return math.huge
	end
	local TargetPos = target:IsA("Model") and target:GetPivot().Position or target.Position
	return (TargetPos - RootPos).Magnitude
end
local function GetDistanceStart(Objective, Start)
	if typeof(Objective) == "Vector3" and typeof(Start) == "Vector3" then
		return (Objective - Start).Magnitude
	elseif typeof(Objective) == "Vector2" and typeof(Start) == "Vector2" then
		return (Objective - Start).Magnitude
	else
		return math.huge
	end
end

local function IsBehindWall(startPos, endPos)
	if not startPos or not endPos then
		return false
	end

	local targets = c().AimAssiantTarget
	if targets then table.insert(ignoreList, targets) end
	rayParams.FilterDescendantsInstances = ignoreList

end

local function solveQuadratic(A, B, C)
	local discriminant = B^2 - 4*A*C
	if discriminant < 0 then
		return nil, nil
	end
	local sqrtDisc = MathSqrt(discriminant)
	local root1 = (-B - sqrtDisc) / (2*A)
	local root2 = (-B + sqrtDisc) / (2*A)
	return root1, root2
end
local function getBallisticFlightTime(direction, gravity, projectileSpeed)
	local root1, root2 = solveQuadratic(
		gravity:Dot(gravity) / 4,
		gravity:Dot(direction) - projectileSpeed^2,
		direction:Dot(direction)
	);
	if root1 and root2 then
		if root1 > 0 and root1 < root2 then
			return MathSqrt(root1);
		elseif root2 > 0 and root2 < root1 then
			return MathSqrt(root2);
		end;
	end;
	return 0;
end;
local function projectileDrop(acceleration, time)
	return 0.5 * acceleration * (time ^ 2)
end
local function PredictPosition(targetPos, approxVel, travelTime, gravity)
	return targetPos + approxVel * travelTime + projectileDrop(gravity,travelTime)
end

		local function getSkeletonPoints(character)
			local parts = {
				"Head",
				"UpperTorso", "LowerTorso", "HumanoidRootPart",
				"LeftUpperArm", "LeftLowerArm", "LeftHand",
				"RightUpperArm", "RightLowerArm", "RightHand",
				"LeftUpperLeg", "LeftLowerLeg", "LeftFoot",
				"RightUpperLeg", "RightLowerLeg", "RightFoot",
			}
			local bones = {}
			for _, partName in pairs(parts) do
				local part = character:FindFirstChild(partName)
				if part and part:IsA("BasePart") then
					bones[partName] = part
				end
			end
			
			if not bones.UpperTorso and bones.HumanoidRootPart then
				bones.UpperTorso = bones.HumanoidRootPart
			end
			if not bones.LowerTorso and bones.HumanoidRootPart then
				bones.LowerTorso = bones.HumanoidRootPart
			end
			return bones
		end

		local function updateSkeletonLines(player, bones, t)
			local data = DrawingsData.Players[player]
			if not data or not data.Skeleton then return end

			local function drawBone(lineName, partA, partB, offsetA, offsetB)
				local line = data.Skeleton[lineName]
				if not line then return end
				if not partA or not partB then
					line.Visible = false
					return
				end
				local posA = partA.Position + (offsetA or Vector3.zero)
				local posB = partB.Position + (offsetB or Vector3.zero)
				local p1, on1 = WorldToViewPoint(posA)
				local p2, on2 = WorldToViewPoint(posB)
				if on1 and on2 then
					setLine(line, Vector2New(p1.X, p1.Y), Vector2New(p2.X, p2.Y), blue, true)
				else
					line.Visible = false
				end
			end

			drawBone("HeadToUpperTorso",     bones.Head,         bones.UpperTorso,       headOffsetY, nil)

			drawBone("RightUpperArmToLowerArm",   bones.RightUpperArm, bones.RightLowerArm, nil, nil)

		end
		local function get3DBoxCorners(root, head)
			local cf   = root.CFrame
			local hw   = 1.5
			local hd   = 1.0
			local yTop = (head.Position.Y + 0.7) - root.Position.Y
			local yBot = -3.2
			local corners3D = {
				cf * Vector3New( hw, yTop,  hd),
				cf * Vector3New(-hw, yTop,  hd),
				cf * Vector3New(-hw, yTop, -hd),
				cf * Vector3New( hw, yTop, -hd),
				cf * Vector3New( hw, yBot,  hd),
				cf * Vector3New(-hw, yBot,  hd),
				cf * Vector3New(-hw, yBot, -hd),
				cf * Vector3New( hw, yBot, -hd),
			}
			local pts2D = {}
			local allOn = true
			for i, p3 in ipairs(corners3D) do
				local sp, on = WorldToViewPoint(p3)
				pts2D[i] = Vector2New(sp.X, sp.Y)
				if not on then allOn = false end
			end
			return pts2D, allOn
		end

		local EMA_PING     = 0.10
		local HISTORY_SIZE = 20
		local MAX_VEL      = 60
		local DEFAULT_PROJ = 900
		local GRAVITY_VEC = Vector3.new(0, -workspace.Gravity, 0)
		local function smoothVelocity(target, rawVel)
			local mag = rawVel.Magnitude
			if mag > MAX_VEL then rawVel = rawVel * (MAX_VEL / mag) end
			SmoothedVel[target] = SmoothedVel[target]
				and SmoothedVel[target] * (1 - EMA_VEL) + rawVel * EMA_VEL
				or rawVel
			return SmoothedVel[target]
		end
		local function smoothAccel(target, rawAccel)
			local mag = rawAccel.Magnitude
			if mag > 150 then rawAccel = rawAccel * (150 / mag) end
			SmoothedAccel[target] = SmoothedAccel[target]

					while #hist > HISTORY_SIZE do table.remove(hist, 1) end
					local rawVel   = Vector3.zero
					local rawAccel = Vector3.zero
					if #hist >= 2 then
						local h1  = hist[#hist]
						local h2  = hist[#hist - 1]
						local dt  = MathMax(h1.time - h2.time, 1e-4)
						rawVel    = (h1.pos - h2.pos) / dt
						if #hist >= 4 then
							local h3  = hist[#hist - 2]
							local h4  = hist[#hist - 3]
							local dt1 = MathMax(h1.time - h2.time, 1e-4)
							local dt2 = MathMax(h2.time - h3.time, 1e-4)
							local dt3 = MathMax(h3.time - h4.time, 1e-4)
							local v1  = (h1.pos - h2.pos) / dt1
							local v2  = (h2.pos - h3.pos) / dt2
							local v3  = (h3.pos - h4.pos) / dt3
							local a1  = (v1 - v2) / MathMax((dt1 + dt2) * 0.5, 1e-4)
							local a2  = (v2 - v3) / MathMax((dt2 + dt3) * 0.5, 1e-4)
							rawAccel  = (a1 + a2) * 0.5
						end
					end
					PredictCache[TargetChar] = {
						sVel   = smoothVelocity(TargetChar, rawVel),
						sAccel = smoothAccel(TargetChar, rawAccel),

				end

					Predict_Position.X,

					Predict_Position.Z
				)

					and ammoType ~= "shotgun"
					and IsBehindWall(ClientPos, Predict_Position)

					args[3] = CFrameNew(math.huge, math.huge, math.huge)
				else
					args[3] = CFrameNew(ClientPos, Predict_Position)
				end
				if args[4] then
					for _, v in pairs(args[4]) do
						for _, x in pairs(v) do
							x.Normal   = Vector3New(0, 1, 0)
							x.Position = Predict_Position
							x.Instance = TargetPart
						end
					end
				end

function func.Snap()
	local BasePosition = Char.get_hrp().Position
	local OffsetY = c().SnapAmount or 1
	while task_wait(0.05) do
		if c().Snap then
			xpcall(function()
				if IsAlive(Character) then
					local CurrentOffset = c().SnapAmount or 1
					if CurrentOffset ~= OffsetY then
						OffsetY = CurrentOffset
					end
					local CurrentY = Char.get_hrp().Position.Y
					local deltaY = GetDeltaY(BasePosition.Y, CurrentY, CurrentOffset)
					Char.get_hrp().CFrame = Char.get_hrp().CFrame * CFrameNew(0, deltaY, 0)
				end
			end,warn)
		end
	end
end
function func.ItemDroppedGraber()
	local ActiveConnections = {}
	while task_wait(0.1) do
		if c().ItemDroppedGraber then
			xpcall(function()
				local Character = Client.Character
				if not Character or not Character.Parent then return end
				local HumanoidRootPart = Character:FindFirstChild("HumanoidRootPart")
				if not HumanoidRootPart then return end
				local CharacterPosition = Character:GetPivot().Position
				local Radius = 45
				
				for item, conn in pairs(ActiveConnections) do
					if not item or not item.Parent then
						conn:Disconnect()
						ActiveConnections[item] = nil
					end
				end
				for _, v in ipairs(DroppedFolder:GetChildren()) do
					if v:IsA('Model') and v:FindFirstChild("PickUpZone") then
						local itemRarity = nil
						pcall(function()
							local itemType = v:GetAttribute("item_type")
							if itemType then
								local itemFolder = ReplicatedStorage:FindFirstChild("Items")
								if itemFolder then
									local typeFolder = itemFolder:FindFirstChild(itemType)
									if typeFolder then
										local itemDef = typeFolder:FindFirstChild(v.Name)
										if itemDef then
											itemRarity = itemDef:GetAttribute("RarityName")
										end
									end
								end
							end
							if not itemRarity then
								itemRarity = v:GetAttribute("RarityName")
							end
						end)
						local blocked = false
						if c().GrabBlacklistEnabled and itemRarity then
							local blacklist = c().GrabBlacklistRarity
							if type(blacklist) == "table" and next(blacklist) then
								
								
								if blacklist[itemRarity] == true then
									blocked = true
								end
							end
						end
						if not blocked then
							local ItemPosition = v:GetPivot().Position
							local Distance = (ItemPosition - CharacterPosition).Magnitude
							if Distance <= Radius then
								local PickUpZone = v:FindFirstChild("PickUpZone")
								if PickUpZone and not ActiveConnections[v] then
									local conn
									conn = RunService.Heartbeat:Connect(function()
										if not v or not v.Parent or not Character or not Character.Parent then
											conn:Disconnect()
											ActiveConnections[v] = nil
											return
										end
										pcall(function()
											firetouchinterest(PickUpZone, HumanoidRootPart, 0)
											firetouchinterest(PickUpZone, HumanoidRootPart, 1)
										end)
									end)
									ActiveConnections[v] = conn
								end
							end
						end
					end
				end
			end,warn)
		else
			
			for item, conn in pairs(ActiveConnections) do
				conn:Disconnect()
				ActiveConnections[item] = nil
			end
		end
	end

			local camPos = center + Vector3New(math.cos(angle) * radius, height, math.sin(angle) * radius)

		end,
		
		function(char, t)
			local root = char:FindFirstChild("HumanoidRootPart")
			if not root then return end
			local center = root.Position + Vector3New(0, 1.5, 0)

			local camPos = center + CFrameAngles(0, angle, 0).LookVector * -dist + Vector3New(0, 2, 0)
			CurrentCamera.CFrame = CFrameLookAt(camPos, center)

			if not root then return end

			local angle = t * 0.9
			local camPos = center + Vector3New(math.cos(angle) * radius, height, math.sin(angle) * radius)
			CurrentCamera.CFrame = CFrameLookAt(camPos, center + Vector3New(0, 1, 0))
		end,
		
		function(char, t)
			local root = char:FindFirstChild("HumanoidRootPart")
			if not root then return end
			local center = root.Position + Vector3New(0, 1.5, 0)
			local side = math.sin(t * 0.8) * 8
			local camPos = center + Vector3New(side, 3, -10)
			CurrentCamera.CFrame = CFrameLookAt(camPos, center)
		end,
	}
	local currentSeq = 1
	local seqTimer = 0
	local SEQ_DURATION = 4
	while task_wait() do
		if not c().AFKCamera then continue end
		local char = Client.Character
		if not char then continue end
		local idle = tick() - lastInputTime
		if idle >= AFK_TIMEOUT and not afkActive then
			afkActive = true
			seqTimer = tick()
			currentSeq = MathRandom(1, #sequences)
			originalCameraType = CurrentCamera.CameraType
			pcall(function()
				CurrentCamera.CameraType = Enum.CameraType.Scriptable
			end)
		end
		if afkActive then
			
			if tick() - seqTimer >= SEQ_DURATION then
				seqTimer = tick()
				currentSeq = (currentSeq % #sequences) + 1
			end
			pcall(function()
				sequences[currentSeq](char, tick())
			end)
		end
	end
	for _, conn in ipairs(inputConns) do conn:Disconnect() end
end
function func.AntiDied()

	while task_wait() do

				local RootPartModule = Char.get_hrp()

				if obj:IsA("BasePart") then dist = distTo(obj.Position)
				elseif obj:IsA('Model') and obj.PrimaryPart then dist = distTo(obj.PrimaryPart.Position) end

			end
		end

function Sf:Teleport(destination, value, t, speed)
	cATM().StopWalking = false
	cATM()['Running'] = true
	local char = Client.Character or Client.CharacterAdded:Wait()
	local RootPart = char:WaitForChild("HumanoidRootPart")
	local Humanoid = char:WaitForChild("Humanoid")
	local path = PathfindingService:CreatePath({
		AgentCanJump = true,
		AgentJumpHeight = 2.5,
		AgentHeight = 8,
		AgentRadius = 2.5,
		AgentMaxSlope = 90,
		Costs = {BlockedNode = 50, DoorArea = 1}
	})
	local success = pcall(function()
		path:ComputeAsync(RootPart.Position, destination)
	end)
	if not success then
		cATM()['Running'] = false
		return
	end
	if path.Status == Enum.PathStatus.Success then
		local waypoints = path:GetWaypoints()
		for _, wp in pairs(waypoints) do
			local offsetY = (wp.Action == Enum.PathWaypointAction.Jump) and 10 or 4
			local targetPos = wp.Position + Vector3.new(0, offsetY, 0)
			local startPos = RootPart.Position
			local dir = (targetPos - startPos).Unit
			local dist = (targetPos - startPos).Magnitude
			local movedDist = 0
			local speed = speed or 30
			local startTime = tick()
			while movedDist < dist and not cATM().StopWalking and value do
				task.wait()
				cATM()['Running'] = true
				if cATM().StopWalking or not value or Humanoid.Health <= 0 then
					cATM()['Running'] = false
					break
				end
				local elapsedTime = tick() - startTime
				movedDist = MathMin(elapsedTime * speed, dist)
				local newPos = startPos + dir * movedDist
				if Humanoid.Sit then
					Humanoid.Sit = false
				end
				RootPart:PivotTo(CFrame.new(newPos))
				Sf:Ac("set_sprinting_1", true)
			end
			cATM()['Running'] = false
			if not value then
				break
			end
		end
	end
	cATM()['Running'] = false
end
local function shouldContinue(value)
	if value == nil then
		return cATM().AutoFarmATM
	end
	return value
end
function Sf:Drive(model, destination, value, t)
	if not model or not model.PrimaryPart then
		cATM()['Running'] = false
		return
	end
	cATM()["StopWalking"] = false
	local path = PathfindingService:CreatePath({
		AgentRadius = cATM().VechineType == "car" and 6 or 3.5,
		AgentHeight = 8,
		AgentCanJump = true,
		AgentMaxSlope = 50,
		AgentCanClimb = false,
		WaypointSpacing = 8,
		Costs = {
			BlockedNode = 100,
			Cars = 1,
			Water = math.huge,
			DoorArea = 1
		}
	})
	local success = pcall(function()
		path:ComputeAsync(model.PrimaryPart.Position, destination)
	end)
	if not success or path.Status ~= Enum.PathStatus.Success then
		cATM()['Running'] = false
		return
	end
	local waypoints = path:GetWaypoints()
	for i, wp in pairs(waypoints) do
		if not model or not model.PrimaryPart then
			cATM()['Running'] = false
			return
		end
		local startPos = model:GetPivot().Position
		local heightOffset = 2
		if wp.Action == Enum.PathWaypointAction.Jump then
			heightOffset = 8
		end
		local goalPos = wp.Position + Vector3.new(0, heightOffset, 0)
		local dir = (goalPos - startPos).Unit
		local dist = (goalPos - startPos).Magnitude
		local movedDist = 0
		local speed = cATM().InstantVechineSpeed or 55
		local startTime = tick()
		while movedDist < dist and shouldContinue(value) do
			task.wait(0)
			if not model or not model.PrimaryPart then
				cATM()['Running'] = false
				return
			end
			cATM()['Running'] = true
			if not shouldContinue(value) then
				cATM()['Running'] = false
				break
			end
			if not Humanoid.Sit then
				cATM()['Running'] = false
				return self:Drive(model, destination, value, t)
			end
			local dt = tick() - startTime
			movedDist = MathMin(dt * speed, dist)
			local newPos = startPos + dir * movedDist
			local lookDirection = dir
			if i < #waypoints then
				local nextWp = waypoints[i + 1]
				lookDirection = (nextWp.Position - wp.Position).Unit
			end
			local Rotation = CFrame.new(newPos) * CFrame.Angles(0, math.atan2(lookDirection.X, lookDirection.Z), 0)
			model:PivotTo(Rotation)
			for _, part in ipairs(model:GetDescendants()) do
				if part:IsA("BasePart") then
					part.AssemblyLinearVelocity = Vector3.zero
					part.AssemblyAngularVelocity = Vector3.zero
				end
			end
		end
		cATM()['Running'] = false
		if not shouldContinue(value) then
			break
		end
	end
	cATM()['Running'] = false
end
local SafeHeight = {
    MinCharacter = 200,
    MinVehicle = 200,
    Emergency = 50,
    RespawnHeight = 260
}
task.spawn(function()
    while task.wait(0) do
        pcall(function()
            if not cATM().AutoFarmATM then return end
            for _, vehicle in pairs(workspace.Vehicles:GetChildren()) do
                if vehicle:GetAttribute('OwnerUserId') == UserId and vehicle.PrimaryPart then
                    local currentY = vehicle.PrimaryPart.Position.Y
                    for _, part in pairs(vehicle:GetDescendants()) do
                        if part:IsA("BasePart") then
                            if part.AssemblyLinearVelocity.Magnitude > 100 then
                                part.AssemblyLinearVelocity = part.AssemblyLinearVelocity.Unit * 100
                            end
                            local verticalVelocity = part.AssemblyLinearVelocity.Y
                            if verticalVelocity > 15 then
                                part.AssemblyLinearVelocity = Vector3.new(
                                    part.AssemblyLinearVelocity.X,
                                    -5,
                                    part.AssemblyLinearVelocity.Z
                                )
                            end
                            if math.abs(part.AssemblyAngularVelocity.X) > 5 or math.abs(part.AssemblyAngularVelocity.Z) > 5 then
                                part.AssemblyAngularVelocity = Vector3.new(0, part.AssemblyAngularVelocity.Y, 0)
                            end
                        end
                    end
                    local downforce = 15
                    local maxHeightAboveGround = 8
                    local rayOrigin = vehicle.PrimaryPart.Position
                    local rayDirection = Vector3.new(0, -100, 0)
                    local raycastParams = RaycastParams.new()
                    raycastParams.FilterDescendantsInstances = {vehicle, Character}
                    raycastParams.FilterType = Enum.RaycastFilterType.Exclude
                    local rayResult = workspace:Raycast(rayOrigin, rayDirection, raycastParams)
                    if rayResult then
                        local distanceToGround = (rayOrigin - rayResult.Position).Magnitude
                        if distanceToGround > maxHeightAboveGround then
                            for _, part in pairs(vehicle:GetDescendants()) do
                                if part:IsA("BasePart") then
                                    part.AssemblyLinearVelocity = Vector3.new(
                                        part.AssemblyLinearVelocity.X,
                                        -downforce,
                                        part.AssemblyLinearVelocity.Z
                                    )
                                end
                            end
                        end
                    end
                    if currentY < SafeHeight.Emergency then
                        vehicle:PivotTo(RootPart.CFrame * CFrame.new(0, 10, 0))
                        for _, part in pairs(vehicle:GetDescendants()) do
                            if part:IsA("BasePart") then
                                part.Velocity = Vector3.zero
                                part.RotVelocity = Vector3.zero
                                part.AssemblyLinearVelocity = Vector3.zero
                                part.AssemblyAngularVelocity = Vector3.zero
                                part.Anchored = true
                                task.wait(0)
                                part.Anchored = false
                            end
                        end
                    elseif currentY < SafeHeight.MinVehicle then
                        local targetPos = Vector3.new(
                            vehicle.PrimaryPart.Position.X,
                            SafeHeight.RespawnHeight,
                            vehicle.PrimaryPart.Position.Z
                        )
                        vehicle:PivotTo(CFrame.new(targetPos) * vehicle:GetPivot().Rotation)
                        for _, part in pairs(vehicle:GetDescendants()) do
                            if part:IsA("BasePart") then
                                part.Velocity = Vector3.zero
                                part.RotVelocity = Vector3.zero
                                part.AssemblyLinearVelocity = Vector3.zero
                                part.AssemblyAngularVelocity = Vector3.zero
                            end
                        end
                    end
                end
            end
        end)
    end
end)

    local keys = {}

local function Tg(def, Tab, Description, Title, Keys, Callback)
    local taskThread
    local toggle = Tab:Toggle({
        Title = Title,
        Icon = "check",
        Type = "Checkbox",
        Desc = Description,
        Value = c()[Keys] or def,
        Color = Color3.fromRGB(255, 80, 80),
        Callback = function(state)
            Config_Manager.S()[Keys] = state
            c()[Keys] = state
            if Callback and typeof(Callback) == "function" then
                Callback(state)
            end
            if not state then
                if taskThread then
                    task.cancel(taskThread)
                    taskThread = nil
                end
            end
            if state then
                if func[Keys] then
                    taskThread = task_spawn(func[Keys])
                end
            end
            Config_Manager.SG()
        end
    })
    if toggle then toggle.Callback(c()[Keys]) end
    return toggle
end

        local cursor = ""
        local best   = nil
        local pages  = 0
        repeat
            local ok, result = pcall(function()
                local url = string.format(
                    "https://games.roblox.com/v1/games/%s/servers/Public?sortOrder=Desc&limit=100&cursor=%s",

                )

            if not ok or not result or not result.data then break end
            for _, srv in ipairs(result.data) do
                local isFull = srv.playing >= srv.maxPlayers
                if not isFull then
                    if not best or srv.playing > best.playing then

            })

            TeleportService:TeleportToPlaceInstance(PlaceId, best.id, Client)
        else
            WindUINew:Notify({
                Title   = "หาของทอง",
                Content = "ไม่พบ server ที่เข้าได้",
                Duration = 4
            })
        end
    end)

local ShiftLockSection = Section(Tabs.Custom, "Shift Lock")
do
    local ShiftLockEnabled = false
    local ShiftLockConn = nil
    local OriginalMouseBehavior = nil
    local function enableShiftLock()
        local cam = workspace.CurrentCamera
        local UserInputService = cloneref(game:GetService("UserInputService"))
        local RunService = cloneref(game:GetService("RunService"))
        OriginalMouseBehavior = UserInputService.MouseBehavior
        ShiftLockConn = RunService.RenderStepped:Connect(function()
            if not Character or not Character:FindFirstChild("HumanoidRootPart") then return end
            local hrp = Character.HumanoidRootPart
            local camCF = cam.CFrame
            local lookVector = Vector3New(camCF.LookVector.X, 0, camCF.LookVector.Z).Unit
            hrp.CFrame = CFrameNew(hrp.Position, hrp.Position + lookVector)
            UserInputService.MouseBehavior = Enum.MouseBehavior.LockCenter
        end)
    end
    local function disableShiftLock()
        if ShiftLockConn then
            ShiftLockConn:Disconnect()
            ShiftLockConn = nil
        end
        pcall(function()
            local UserInputService = cloneref(game:GetService("UserInputService"))
            UserInputService.MouseBehavior = OriginalMouseBehavior or Enum.MouseBehavior.Default
        end)
    end
    Tg(false, ShiftLockSection, "เปิด Shift Lock", "ล็อคกล้องตามทิศที่มอง", "ShiftLockToggle", function(State)
        ShiftLockEnabled = State
        if State then
            enableShiftLock()
            WindUINew:Notify({ Title = "Shift Lock", Content = "เปิดใช้งาน Shift Lock แล้ว", Duration = 2 })
        else
            disableShiftLock()
            WindUINew:Notify({ Title = "Shift Lock", Content = "ปิด Shift Lock แล้ว", Duration = 2 })
        end
    end)
    local ShiftLockKeyConn = nil
    local ShiftLockKeyEnabled = false
    local function startKeyListener()
        if ShiftLockKeyConn then return end
        local UserInputService = cloneref(game:GetService("UserInputService"))
        ShiftLockKeyConn = UserInputService.InputBegan:Connect(function(input, gpe)
            if gpe then return end
            if input.KeyCode == Enum.KeyCode.LeftShift then
                ShiftLockKeyEnabled = not ShiftLockKeyEnabled
                if ShiftLockKeyEnabled then
                    enableShiftLock()
                else
                    disableShiftLock()
                end
            end
        end)
    end
    local function stopKeyListener()
        if ShiftLockKeyConn then
            ShiftLockKeyConn:Disconnect()
            ShiftLockKeyConn = nil
        end
        if ShiftLockKeyEnabled then
            disableShiftLock()
            ShiftLockKeyEnabled = false
        end
    end
    Tg(false, ShiftLockSection, "กด Shift เพื่อสลับ", "กด Left Shift เพื่อเปิด/ปิด Shift Lock", "ShiftKeyToggle", function(State)
        if State then
            startKeyListener()
            WindUINew:Notify({ Title = "Shift Lock", Content = "กด Left Shift เพื่อสลับ", Duration = 2 })
        else
            stopKeyListener()
        end
    end)
end
