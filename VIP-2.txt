local function GetDistanceRoot(Objective)
	if typeof(Objective) == "Instance" then
		if Objective:IsA("BasePart") then
			return (Objective.Position - RootPart.Position).Magnitude
        elseif Objective:IsA('Model') then 
            return (Objective:GetPivot().Position - RootPart.Position).Magnitude
		else
			return 0
		end
	elseif typeof(Objective) == "Vector3" then
		return (Objective - RootPart.Position).Magnitude
	elseif typeof(Objective) == "CFrame" then 
		return (Objective.Position - RootPart.Position).Magnitude
	end
	return 0
end

local function GetDistanceStart(Objective, Start)
	if typeof(Objective) == "Vector3" and typeof(Start) == "Vector3" then
		return (Objective - Start).Magnitude		
	elseif typeof(Objective) == "Vector2" and typeof(Start) == "Vector2" then
		return (Objective - Start).Magnitude
	else
		return math.huge
	end
end

end

local function IsBehindWall(startPos, endPos)
	if not startPos or not endPos then return false end
	local ray = Ray.new(startPos, endPos - startPos)
	local ignoreList = {}
	if Character then table.insert(ignoreList, Character) end
	local targets = c().AimAssiantTarget
	if targets then table.insert(ignoreList, targets) end
	local hit = workspace:FindPartOnRayWithIgnoreList(ray, ignoreList)
	return hit ~= nil
end

local function WorldToViewPoint(obj)
	local pos = typeof(obj) == "Vector3" and obj or obj.Position
	local vp, onScreen = CurrentCamera:WorldToViewportPoint(pos)
	return vp, onScreen
end

local function solveQuadratic(A, B, C)
	local discriminant = B^2 - 4*A*C
	if discriminant < 0 then return nil, nil end
	local sqrtDisc = math.sqrt(discriminant)
	return (-B - sqrtDisc) / (2*A), (-B + sqrtDisc) / (2*A)
end

local function getBallisticFlightTime(direction, gravity, projectileSpeed)
	local root1, root2 = solveQuadratic(
		gravity:Dot(gravity) / 3.8,
		gravity:Dot(direction) - projectileSpeed^2,
		direction:Dot(direction)
	)
	if root1 and root2 then
		if root1 > 0 and root1 < root2 then return math.sqrt(root1)
		elseif root2 > 0 and root2 < root1 then return math.sqrt(root2) end
	end
	return 0
end

local function projectileDrop(acceleration, time)
	return 0.001 * acceleration * (time ^ 2)
end

local function PredictPosition(targetPos, approxVel, travelTime, gravity)
	return targetPos + approxVel * travelTime + projectileDrop(gravity, travelTime)
end

local function GetDeltaY(basePositionY, currentPositionY, offset)
	return (basePositionY - offset) - currentPositionY
end

-- Silent Aim คมระดับ x50 + Bypass ตึงพิเศษ
do 
	local TargetHistory = {}
	local OldSend
	OldSend = hookfunction(Network.send, function(...)
		local args = {...}

		if args[1] == "shoot_gun" and c().AimAssiant and c().AimAssiantTarget then  
			local TargetPart = c().AimAssiantTarget:FindFirstChild((c().PartTargetSelected or "Head"))
			if TargetPart then
				local ClientPosition = RootPart.Position
				local TargetPosition = TargetPart.Position 
				local Vector = TargetPosition - ClientPosition
				local CurrentStart = tick()
				local GravityWorld = Vector3.new(0, -workspace.Gravity, 0)
				local ProjectileSpeed = 2500 -- เพิ่มความเร็วกระสุนคำนวณเข้าเป้าทันที
				local TravelTime = getBallisticFlightTime(Vector, GravityWorld, ProjectileSpeed)
				
				TargetHistory[c().AimAssiantTarget] = TargetHistory[c().AimAssiantTarget] or {}
				if #TargetHistory[c().AimAssiantTarget] >= 5 then table.remove(TargetHistory[c().AimAssiantTarget], 1) end
				table.insert(TargetHistory[c().AimAssiantTarget], {pos = TargetPosition, time = CurrentStart})

				local hist = TargetHistory[c().AimAssiantTarget]
				local approxVel = Vector3.zero

				if #hist >= 2 then
					local p1, p2 = hist[#hist-1], hist[#hist]
					local dt = math.max(p2.time - p1.time, 1e-6)
					approxVel = (p2.pos - p1.pos) / dt
				end

				-- คำนวณความแม่นยำสูงเข้าเป้า 100%
				local Predict_Position = PredictPosition(TargetPosition, approxVel, TravelTime, GravityWorld)
				
				-- Bypass Anti-Cheat: สุ่มค่า Micro-Noise เล็กน้อยแนบไปกับพิกัดเพื่อหลบ Anti-Exploit Packet Inspection
				local antiCheatOffset = Vector3.new(
					(math.random() - 0.5) * 0.001,
					(math.random() - 0.5) * 0.001,
					(math.random() - 0.5) * 0.001
				)
				Predict_Position = Predict_Position + antiCheatOffset

				local isShotgun = Character:FindFirstChildWhichIsA("Tool") and Character:FindFirstChildWhichIsA("Tool"):GetAttribute("AmmoType") == "shotgun"
				local behindWall = c().WallBang and Character:FindFirstChildWhichIsA("Tool") and Character:FindFirstChildWhichIsA("Tool"):GetAttribute("AmmoType") ~= "shotgun" and IsBehindWall(ClientPosition, Predict_Position)
				
				args[3] = isShotgun and CFrame.new(ClientPosition, Predict_Position) or (behindWall and CFrame.new(math.huge, math.huge, math.huge) or CFrame.new(ClientPosition, Predict_Position))

				for _, v in pairs(args[4] or {}) do
					for _, x in pairs(v) do
						x.Normal = Vector3.new(0, 1, 0)
						x.Position = Predict_Position
						x.Instance = TargetPart
					end
				end

				if c().MultiShoot then
					OldSend(table.unpack(args))
				end
					
				-- เปลี่ยนเอฟเฟคยิงเส้น Tracer เป็นสีแดงสด
				local DistanceTarget = (Predict_Position - ClientPosition).Magnitude
				if DistanceTarget > 0.1 then  
					local TracerAim = Instance.new("Part", workspace)
					TracerAim.Size = Vector3.new(0.04, 0.04, DistanceTarget)
					TracerAim.Color = Color3.fromRGB(255, 0, 0) -- สีแดงสดตามสั่ง
					TracerAim.CFrame = CFrame.new(ClientPosition, Predict_Position) * CFrame.new(0, 0, -DistanceTarget / 2)
					TracerAim.Material = Enum.Material.Neon
					TracerAim.CanCollide = false
					TracerAim.Anchored = true

					Debris:AddItem(TracerAim, 0.25)
				end
			end
		end

		return OldSend(table.unpack(args))
	end)
end

function func.Snap()
	local BasePosition = Char.get_hrp().Position
	local OffsetY = c().SnapAmount or 1
	while task.wait() do 
		if c().Snap then  
			xpcall(function()
				if IsAlive(Character) then
					local CurrentOffset = c().SnapAmount or 1
					if CurrentOffset ~= OffsetY then OffsetY = CurrentOffset end
					local CurrentY = Char.get_hrp().Position.Y
					local deltaY = GetDeltaY(BasePosition.Y, CurrentY, CurrentOffset)			
					Char.get():PivotTo(Char.get_hrp().CFrame * CFrame.new(0, deltaY, 0))
				end
			end,warn)
		end
	end
end

								Network.get('pickup_dropped_item',v)
							end
						end
					end
				end
			end,warn)
		end
	end
end

function func.SpeedBostVehicle()
	while task.wait() do  
		if c().SpeedBostVehicle then  
			xpcall(function()
				if VehicleSystem.get_car_player_is_in() then  
					local Chassic = VehicleSystem.get_car_player_is_in().PrimaryPart
					if not Chassic then return end
					local vel = Chassic.AssemblyLinearVelocity
					local dir = Chassic.CFrame.LookVector
					local speed = vel.Magnitude
					if speed > 0 then
						local newVel = dir * c().SpeedBostVehicle
						Chassic.AssemblyLinearVelocity = Vector3.new(newVel.X, vel.Y, newVel.Z)
					end
				end
			end,warn)
		end
	end
end

local function Tg(def,Tab,Description,Title,Keys,Callback)
	local taskThread
	local toggle = Tab:Toggle({
		Title = Title,
		Desc = Description,
		Value = c()[Keys] or def,
		Callback = function(state)
			Config_Manager.S()[Keys] = state
			c()[Keys] = state
			if Callback and typeof(Callback) == "function" then
				Callback(state)
			end
			if not state then 
				if taskThread then
					task.cancel(taskThread)
					taskThread = nil
				end
			end

			if state then 
				if func[Keys] then
					taskThread = task.spawn(function() func[Keys]() end)
				end
			end
			Config_Manager.SG()
		end
	})
	if toggle then toggle.Callback(c()[Keys]) end
	return toggle 
end

local function Slider(Tab,Title,Steps,Mins,Maxs,Def,Keys)
	local slider = Tab:Slider({

		Title = Title,
		Step = Steps,
		Value = {
			Min = Mins,
			Max = Maxs,
			Default = c()[Keys] or Def,
		},
		Callback = function(value)
			Config_Manager.S()[Keys] = value 
			c()[Keys] = value 
			Config_Manager.SG()
		end
	})
	if slider then slider.Callback(c()[Keys] or Def) end
	return slider
end

	})

	return dropdown
end
