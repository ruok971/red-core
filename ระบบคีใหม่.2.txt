-- แจ้งเตือนเมื่อรันสำเร็จ
game:GetService("StarterGui"):SetCore("SendNotification", {
    Title = "สนับสนุนโดยคนรวยแถมโคยใหญ่",
    Text = "อัพเดทเพิ่ม FOV + เส้นระบุเป้าหมาย",
    Duration = 5
})

local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local Workspace = game:GetService("Workspace")
local LocalPlayer = Players.LocalPlayer
local Camera = Workspace.CurrentCamera
local Mouse = LocalPlayer:GetMouse()

getgenv().SilentAim = true
getgenv().CurrentTarget = nil

-- Configuration FOV & Visuals
local FOV_RADIUS = 125
local SIDES = 8

-- สร้างวัตถุ Drawing สำหรับ FOV และ เส้น Tracer
local HexagonLines = {}
for i = 1, SIDES do
    local line = Drawing.new("Line")
    line.Thickness = 2
    line.Transparency = 1
    line.Visible = false
    HexagonLines[i] = line
end

local TargetLine = Drawing.new("Line")
TargetLine.Thickness = 1.5
TargetLine.Color = Color3.fromRGB(255, 0, 0)
TargetLine.Transparency = 1
TargetLine.Visible = false

-- ระบบเลือกเป้าหมาย "หัว" ที่ใกล้กลางจอมากที่สุด และต้องอยู่ในวง FOV ขนาด 90 เท่านั้น
local function GetClosestTarget()
    local closestTarget = nil
    local shortestDistance = math.huge
    local screenCenter = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character and player.Character:FindFirstChild("Head") then
            local humanoid = player.Character:FindFirstChildOfClass("Humanoid")
            if humanoid and humanoid.Health > 0 then
                local head = player.Character.Head
                local distance = (head.Position - Camera.CFrame.Position).Magnitude
                local screenPoint, onScreen = Camera:WorldToViewportPoint(head.Position)
                
                if onScreen then
                    local targetScreenPos = Vector2.new(screenPoint.X, screenPoint.Y)
                    local centerVector = (screenCenter - targetScreenPos).Magnitude
                    
                    if centerVector <= FOV_RADIUS and distance < shortestDistance then
                        shortestDistance = distance
                        closestTarget = head
                    end
                end
            end
        end
    end

    return closestTarget
end

-- Render loop สำหรับอัปเดตตำแหน่ง FOV และ Tracer
local hueBase = 0
RunService.RenderStepped:Connect(function()
    hueBase = (hueBase + 0.003) % 1
    
    local screenCenter = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)

    local points = {}
    for i = 1, SIDES do
        local angle = math.rad((i - 1) * (360 / SIDES))
        local x = screenCenter.X + FOV_RADIUS * math.cos(angle)
        local y = screenCenter.Y + FOV_RADIUS * math.sin(angle)
        table.insert(points, Vector2.new(x, y))
    end

    for i = 1, SIDES do
        local nextIndex = (i % SIDES) + 1
        local line = HexagonLines[i]
        local lineHue = (hueBase + (i / SIDES)) % 1
        local lineRainbowColor = Color3.fromHSV(lineHue, 1, 1)

        line.From = points[i]
        line.To = points[nextIndex]
        line.Color = lineRainbowColor
        line.Visible = getgenv().SilentAim
    end

    if getgenv().SilentAim then
        getgenv().CurrentTarget = GetClosestTarget()
    else
        getgenv().CurrentTarget = nil
    end

    if getgenv().SilentAim and getgenv().CurrentTarget then
        local screenPoint, onScreen = Camera:WorldToViewportPoint(getgenv().CurrentTarget.Position)
        if onScreen then
            TargetLine.From = screenCenter
            TargetLine.To = Vector2.new(screenPoint.X, screenPoint.Y)
            TargetLine.Visible = true
        else
            TargetLine.Visible = false
        end
    else
        TargetLine.Visible = false
    end
end)

-- Metatable Hooking แบบเนียน (Bypass Server Verification)
local mt = getrawmetatable(game)
local oldNamecall = mt.__namecall
local oldIndex = mt.__index

setreadonly(mt, false)

mt.__index = newcclosure(function(self, idx)
    if getgenv().SilentAim and getgenv().CurrentTarget and not checkcaller() and self == Mouse then
        if idx == "Hit" then 
            return getgenv().CurrentTarget.CFrame 
        end
        if idx == "Target" then 
            return getgenv().CurrentTarget 
        end
        if idx == "X" or idx == "Y" then
            local screenPoint = Camera:WorldToViewportPoint(getgenv().CurrentTarget.Position)
            return idx == "X" and screenPoint.X or screenPoint.Y
        end
    end
    return oldIndex(self, idx)
end)

mt.__namecall = newcclosure(function(self, ...)
    local method = getnamecallmethod()
    local args = {...}

    if getgenv().SilentAim and getgenv().CurrentTarget and not checkcaller() then
        local targetHead = getgenv().CurrentTarget

        -- แก้ไขเฉพาะ Raycast ที่ส่งออกจากตัวผู้เล่น เพื่อไม่ให้ขัดกับ Anti-Cheat ฝั่ง Server
        if method == "Raycast" and self == Workspace then
            local origin = args[1]
            if typeof(origin) == "Vector3" then
                local direction = (targetHead.Position - origin)
                args[2] = direction
                return oldNamecall(self, unpack(args))
            end
        end

        -- ส่งค่าตำแหน่งเป้าหมายไปยัง RemoteEvent โดยไม่แก้ไขโครงสร้าง Table ทั้งหมด
        if method == "FireServer" or method == "InvokeServer" then
            for i, arg in ipairs(args) do
                if typeof(arg) == "Vector3" then
                    args[i] = targetHead.Position
                    break
                elseif typeof(arg) == "CFrame" then
                    args[i] = targetHead.CFrame
                    break
                end
            end
        end
    end

    return oldNamecall(self, unpack(args))
end)

setreadonly(mt, true)
