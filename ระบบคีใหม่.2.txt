local function createInstance(class, props)
    local obj = Instance.new(class)
    if props then
        for key, value in pairs(props) do
            obj[key] = value
        end
    end
    return obj
end

local KeyDatabase = {}

local function LoadData()
    if isfile and isfile(DATA_FILE) then
        local success, result = pcall(function()
            return HttpService:JSONDecode(readfile(DATA_FILE))
        end)
        if success and typeof(result) == "table" then
            KeyDatabase = result
        end
    else
        KeyDatabase = {
            ["VIP-SUCCESS-KAISERHUB-KEY-8872-PRO"] = {
                DurationSeconds = 3 * 86400,
                CreatedAt = os.time(),
                AllowedPlayer = "" -- เว้นว่างไว้ = ให้ทุกเครื่อง/ทุกคนใช้ได้
            }
        }
    end
end

local function SaveData()
    if writefile then
        pcall(function()
            writefile(DATA_FILE, HttpService:JSONEncode(KeyDatabase))
        end)
    end
end

LoadData()

local function CheckKeyStatus(key, playerName)
    local data = KeyDatabase[key]
    if not data then return "NOT_FOUND", 0 end
    
    -- ตรวจสอบการล็อกชื่อผู้เล่น
    if data.AllowedPlayer and data.AllowedPlayer ~= "" then
        if string.lower(data.AllowedPlayer) ~= string.lower(playerName) then
            return "USER_MISMATCH", 0
        end
    end

    local expireTime = data.CreatedAt + data.DurationSeconds
    local timeLeft = expireTime - os.time()
    return (timeLeft <= 0) and "EXPIRED" or "VALID", math.max(0, timeLeft)
end

local function RefreshView()
    for _, child in ipairs(ScrollList:GetChildren()) do
        if child:IsA("Frame") then child:Destroy() end
    end

    local count = 0
    for k, v in pairs(KeyDatabase) do
        count = count + 1
        local status, timeLeft = CheckKeyStatus(k, LocalPlayer.Name)
        local hoursLeft = math.floor(timeLeft / 3600)
        local minsLeft = math.floor((timeLeft % 3600) / 60)
        local userTarget = (v.AllowedPlayer and v.AllowedPlayer ~= "") and (" | ผู้ใช้: " .. v.AllowedPlayer) or " | ผู้ใช้: ทุกคน"

        local ItemFrame = createInstance("Frame", {
            Parent = ScrollList, Size = UDim2.new(1, -8, 0, 36),
            BackgroundColor3 = (SelectedKey == k) and Color3.fromRGB(35, 45, 60) or COLORS.BG_CARD,
            BorderSizePixel = 0, ZIndex = 3,
        })
        createInstance("UICorner", { CornerRadius = UDim.new(0, 8), Parent = ItemFrame })

        createInstance("UIStroke", {
            Thickness = 1, Color = (SelectedKey == k) and COLORS.PRIMARY or COLORS.BORDER, Parent = ItemFrame,
        })

        local SelectBtn = createInstance("TextButton", {
            Parent = ItemFrame, BackgroundTransparency = 1, Position = UDim2.new(0, 10, 0, 0),
            Size = UDim2.new(0.82, 0, 1, 0), Font = Enum.Font.Code,
            TextColor3 = (status == "EXPIRED") and COLORS.ERROR or COLORS.SUCCESS, TextSize = 10,
            Text = string.format("[%s] Key: %s%s | เหลือ: %dชม.%dน.", (status == "EXPIRED" and "EXPIRED" or "VALID"), k, userTarget, hoursLeft, minsLeft),
            TextXAlignment = Enum.TextXAlignment.Left, ZIndex = 4,
        })

        SelectBtn.MouseButton1Click:Connect(function()
            SelectedKey = k
            RefreshView()
        end)

        local DelBtn = createInstance("TextButton", {
            Parent = ItemFrame, BackgroundColor3 = COLORS.ERROR, Position = UDim2.new(0.88, 0, 0.15, 0),
            Size = UDim2.new(0.1, 0, 0.7, 0), Font = Enum.Font.GothamBold, Text = "-", TextColor3 = COLORS.TEXT_WHITE, TextSize = 14, ZIndex = 4,
        })
        createInstance("UICorner", { CornerRadius = UDim.new(0, 6), Parent = DelBtn })

        DelBtn.MouseButton1Click:Connect(function()
            KeyDatabase[k] = nil
            if SelectedKey == k then SelectedKey = nil end
            SaveData() RefreshView()
        end)
    end
    ScrollList.CanvasSize = UDim2.new(0, 0, 0, count * 42)
end
