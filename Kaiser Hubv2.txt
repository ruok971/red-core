-- ─── Utility Functions ─────────────────────────────────────
local function GetPing()
    local gui = LocalPlayer:FindFirstChild("PlayerGui")
    if not gui then return 0.2 end
    local stats = gui:FindFirstChild("NetworkStats")
    if not stats then return 0.2 end
    local label = stats:FindFirstChild("PingLabel")
    if not label then return 0.2 end
    local text = label.Text
    if typeof(text) ~= "string" then return 0.2 end
    local ping = tonumber(text:match("%d+"))
    if not ping or ping < 0 or ping > 2000 then return 0.2 end
    return ping / 1000
end

local function IsFriend(playerName)
    for _, friend in ipairs(FriendsList) do
        if friend ~= "" and string.find(string.lower(playerName), string.lower(friend)) then
            return true
        end
    end
    return false
end

local function IsBehindWall(pos1, pos2)
    if not pos1 or not pos2 then return false end
    local direction = pos2 - pos1
    if direction.Magnitude < 1 then return false end
    local rayParams = RaycastParams.new()
    rayParams.FilterDescendantsInstances = {Character}
    rayParams.FilterType = Enum.RaycastFilterType.Exclude
    local result = Workspace:Raycast(pos1, direction, rayParams)
    if result then
        return not (result.Instance and (result.Instance.Parent == Character or result.Instance.Parent == LocalPlayer.Character))
    end
    return false
end

local function PredictPosition(part, velocity)
    if not part then return Vector3.zero end
    local ping = GetPing()
    if ping > 1 then ping = 0.2 end
    local vel = velocity or (part.AssemblyLinearVelocity or Vector3.zero)
    return part.Position + (vel * ping * 1.21)
end

local function GetClosestTarget()
    local closest = nil
    local closestDist = FOVRadius
    local center = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            local targetChar = player.Character
            local head = targetChar:FindFirstChild("Head")
            local humanoid = targetChar:FindFirstChild("Humanoid")
            local torso = targetChar:FindFirstChild("HumanoidRootPart")

            if head and humanoid and humanoid.Health > 0 and torso then
                -- เลือก part ตาม SelectedAimPart
                local aimPart = nil
                if SelectedAimPart == "Head" then
                    aimPart = head
                elseif SelectedAimPart == "Torso" then
                    aimPart = torso
                else -- Both: ใช้ head เป็นหลัก แต่เช็ค torso ด้วยถ้า head บัง
                    aimPart = head
                end

                local screenPos, onScreen = Camera:WorldToViewportPoint(aimPart.Position)
                if onScreen then
                    local screenVec = Vector2.new(screenPos.X, screenPos.Y)
                    local dist = (screenVec - center).Magnitude
                    if dist <= closestDist and not IsFriend(player.Name) then
                        -- เช็ค wallbang ถ้าเปิดใช้งาน
                        if WallbangEnabled or not IsBehindWall(HRP.Position, aimPart.Position) then
                            closestDist = dist
                            closest = player
                        end
                    end
                end
            end
        end
    end
    return closest
end

-- ─── ESP Functions ─────────────────────────────────────────
local function CreateESP(player)
    if ESPDrawings[player] then return end

    local nameText = Drawing.new("Text")
    nameText.Size = 16
    nameText.Center = true
    nameText.Outline = true
    nameText.Color = IsFriend(player.Name) and Color3.fromRGB(0, 255, 0) or Color3.fromRGB(255, 255, 255)
    nameText.Font = 4

    local healthText = Drawing.new("Text")
    healthText.Size = 14
    healthText.Center = true
    healthText.Outline = true
    healthText.Color = Color3.fromRGB(255, 255, 255)
    healthText.Font = 4

    local distText = Drawing.new("Text")
    distText.Size = 14
    distText.Center = true
    distText.Outline = true
    distText.Color = Color3.fromRGB(200, 200, 200)
    distText.Font = 4

    local box = Drawing.new("Square")
    box.Filled = false
    box.Thickness = 1
    box.Color = Color3.fromRGB(255, 255, 255)
    box.Transparency = 0.7
    box.Visible = false

    local healthBar = Drawing.new("Square")
    healthBar.Filled = true
    healthBar.Transparency = 0.6
    healthBar.Visible = false

    local conn = RunService.RenderStepped:Connect(function()
        if not player or not player.Character or not player.Character:FindFirstChild("HumanoidRootPart") then
            for _, d in ipairs({nameText, healthText, distText, box, healthBar}) do
                d.Visible = false
            end
            return
        end

        local targetHRP = player.Character.HumanoidRootPart
        local targetHumanoid = player.Character:FindFirstChild("Humanoid")
        local distance = 0
        if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
            distance = (targetHRP.Position - LocalPlayer.Character.HumanoidRootPart.Position).Magnitude
        end

        local screenPos, onScreen = Camera:WorldToViewportPoint(targetHRP.Position)
        if not onScreen or screenPos.Z <= 0 then
            for _, d in ipairs({nameText, healthText, distText, box, healthBar}) do
                d.Visible = false
            end
            return
        end

        local x, y = screenPos.X, screenPos.Y - 15

        -- Health Bar
        if HealthESP and targetHumanoid and targetHumanoid.Health > 0 then
            local hpRatio = targetHumanoid.Health / (targetHumanoid.MaxHealth > 0 and targetHumanoid.MaxHealth or 1)
            local barW, barH = 4, 60
            local barX = x - barH / 2
            box.Position = Vector2.new(barX, y - barW - 2)
            box.Size = Vector2.new(barH, barW)
            box.Visible = true
            healthBar.Position = Vector2.new(barX, y - barW - 2)
            healthBar.Size = Vector2.new(barH * hpRatio, barW)
            healthBar.Color = Color3.fromHSV(hpRatio * 0.333, 0.8, 0.9)
            healthBar.Visible = true
            y = y - barW - 8
        else
            box.Visible = false
            healthBar.Visible = false
        end

        -- Name
        if NameESP then
            nameText.Text = player.Name
            nameText.Color = IsFriend(player.Name) and Color3.fromRGB(0, 255, 0) or Color3.fromRGB(255, 255, 255)
            nameText.Position = Vector2.new(x, y - 16)
            nameText.Visible = true
        else
            nameText.Visible = false
        end

        -- Distance
        if DistanceESP then
            distText.Text = string.format("%.0f studs", distance)
            distText.Position = Vector2.new(x, screenPos.Y + 20)
            distText.Visible = true
        else
            distText.Visible = false
        end
    end)

    ESPDrawings[player] = { conn = conn, drawings = {nameText, healthText, distText, box, healthBar} }
end

local function LoadESP()
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and not ESPDrawings[player] then
            CreateESP(player)
        end
    end
    Players.PlayerAdded:Connect(function(player)
        if player ~= LocalPlayer then
            player.CharacterAdded:Connect(function()
                task.wait(0.1)
                if not ESPDrawings[player] then
                    CreateESP(player)
                end
            end)
            if player.Character and not ESPDrawings[player] then
                task.wait(0.1)
                CreateESP(player)
            end
        end
    end)
    Players.PlayerRemoving:Connect(function(player)
        if ESPDrawings[player] then
            for _, d in ipairs(ESPDrawings[player].drawings) do
                if d and d.Destroy then pcall(function() d:Destroy() end) end
            end
            if ESPDrawings[player].conn then pcall(function() ESPDrawings[player].conn:Disconnect() end) end
            ESPDrawings[player] = nil
        end
    end)
end

-- ─── Aimbot Render ─────────────────────────────────────────
local AimSmooth = Vector3.new()
RunService.RenderStepped:Connect(function()
    pcall(function()
        ClosestTarget = nil
        if SilentAimEnabled or RedLineEnabled then
            ClosestTarget = GetClosestTarget()
        end

        -- FOV Circle
        if FOVCircle then
            FOVCircle.Visible = SilentAimEnabled or RedLineEnabled
            if SilentAimEnabled or RedLineEnabled then
                if IsTouch then
                    FOVCircle.Position = UDim2.fromScale(0.5, 0.5)
                    FOVCircle.Size = UDim2.fromOffset(FOVRadius * 2, FOVRadius * 2)
                else
                    FOVCircle.Position = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
                    FOVCircle.Radius = FOVRadius
                end
            end
        end

        -- Aim Line / Tracer
        if ClosestTarget and ClosestTarget.Character then
            local targetChar = ClosestTarget.Character
            local targetHumanoid = targetChar:FindFirstChild("Humanoid")
            local aimPart = nil

            if SelectedAimPart == "Head" then
                aimPart = targetChar:FindFirstChild("Head")
            elseif SelectedAimPart == "Torso" then
                aimPart = targetChar:FindFirstChild("HumanoidRootPart")
            else -- Both
                aimPart = targetChar:FindFirstChild("Head")
                if not aimPart then
                    aimPart = targetChar:FindFirstChild("HumanoidRootPart")
                end
            end

            if targetHumanoid and targetHumanoid.Health > 0 and aimPart then
                local center = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
                local predictedPos = PredictPosition(aimPart, targetChar:FindFirstChild("HumanoidRootPart"))
                AimSmooth = AimSmooth:Lerp(predictedPos, 0.75)
                local finalPos = AimSmooth

                local screenPos, onScreen = Camera:WorldToViewportPoint(finalPos)
                if onScreen then
                    -- Red Line
                    if RedLineEnabled then
                        AimLine.Visible = true
                        AimLine.From = center
                        AimLine.To = Vector2.new(screenPos.X, screenPos.Y)
                        AimLine.Color = Color3.fromRGB(255, 50, 50)
                        AimLine.Thickness = 1.5
                    else
                        AimLine.Visible = false
                    end

                    -- Box Tracer ESP
                    if SilentAimEnabled then
                        local headScreenPos = Camera:WorldToViewportPoint(aimPart.Position)
                        local headUp = Camera:WorldToViewportPoint(aimPart.Position + Vector3.new(0, 0.5, 0))
                        local headDown = Camera:WorldToViewportPoint(aimPart.Position - Vector3.new(0, 0.5, 0))
                        local boxSize = math.clamp((headUp - headDown).Magnitude / 2, 8, 25)
                        local halfBox = boxSize

                        local tl = Vector2.new(headScreenPos.X, headScreenPos.Y - boxSize)
                        local tr = Vector2.new(headScreenPos.X + halfBox, headScreenPos.Y - boxSize)
                        local bl = Vector2.new(headScreenPos.X, headScreenPos.Y + boxSize)
                        local br = Vector2.new(headScreenPos.X + halfBox, headScreenPos.Y + boxSize)

                        -- Simple box
                        local boxLines = {}
                        for i = 1, 4 do
                            if not boxLines[i] then boxLines[i] = Drawing.new("Line") end
                            boxLines[i].Color = Color3.fromRGB(255, 255, 255)
                            boxLines[i].Thickness = 1.2
                            boxLines[i].Visible = true
                        end
                        boxLines[1].From, boxLines[1].To = tl, tr
                        boxLines[2].From, boxLines[2].To = tr, br
                        boxLines[3].From, boxLines[3].To = br, bl
                        boxLines[4].From, boxLines[4].To = bl, tl
                    end
                else
                    AimLine.Visible = false
                end
            else
                AimLine.Visible = false
            end
        else
            AimLine.Visible = false
            AimSmooth = Vector3.new()
        end
    end)
end)

-- ─── Silent Aim Hook ───────────────────────────────────────
if SendRemote and SendRemote.FireServer then
    local OriginalFireServer
    pcall(function()
        OriginalFireServer = hookfunction(SendRemote.FireServer, function(self, ...)
            if self ~= SendRemote then
                return OriginalFireServer(self, ...)
            end

            local args = {...}

            if SilentAimEnabled and args[2] == "shoot_gun" and ClosestTarget then
                local targetChar = ClosestTarget.Character
                local targetHead = targetChar and targetChar:FindFirstChild("Head")
                local targetHRP = targetChar and targetChar:FindFirstChild("HumanoidRootPart")
                local targetHumanoid = targetChar and targetChar:FindFirstChild("Humanoid")

                if targetHead and targetHRP and targetHumanoid then
                    local predictedPos = PredictPosition(targetHead, targetHRP)
                    local localHead = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Head")
     
                    local localPos = localHead and localHead.Position or nil

                    -- Wallbang check
                    if WallbangEnabled or not IsBehindWall(localPos, predictedPos) then
                        -- แก้ CFrame ให้เล็งไปที่ predicted position
                        args[4] = CFrame.new(localPos, predictedPos)

                        -- สร้าง hit data สำหรับปืน

                            for i = 1, 6 do
                                local spread = Vector3.new(
                                    math.random(-2, 2) * 0.03,
                                    math.random(-2, 2) * 0.03,
                                    math.random(-2, 2) * 0.03
                                )
                                table.insert(hitData, {
                                    [1] = {
                                        Instance = targetHead,
                                        Normal = Vector3.new(0, 1, 0),
                                        Position = predictedPos + spread
                                    }

                            end
                        else
                            hitData = {
                                [1] = {
                                    [1] = {
                                        Instance = targetHead,
                                        Normal = Vector3.new(0, 1, 0),
                                        Position = predictedPos
                                    }
                                }
                            }
                        end
                        args[5] = hitData

                        -- Visual tracer (red neon part)
                        if localPos then
                            local tracer = Instance.new("Part")
                            tracer.Anchored = true
                            tracer.CanCollide = false
                            tracer.Size = Vector3.new(0.08, 0.08, (predictedPos - localPos).Magnitude)

                            tracer.Material = Enum.Material.Neon
                            tracer.Transparency = 0.35
                            tracer.Color = Color3.fromRGB(255, 0, 0)
                            tracer.Parent = Workspace
                            Debris:AddItem(tracer, 4)

                            -- Check if hit landed
                            local oldHealth = targetHumanoid.Health
                            task.spawn(function()
                                task.wait(0.15)
                                if targetHumanoid and targetHumanoid.Health < oldHealth then
                                    tracer.Color = Color3.fromRGB(0, 255, 0)
                                    -- Hit confirmation particles
                                    local hitMarker = Instance.new("Part")
                                    hitMarker.Size = Vector3.new(0.2, 0.2, 0.2)
                                    hitMarker.Shape = Enum.PartType.Ball
                                    hitMarker.Material = Enum.Material.Neon
                                    hitMarker.Color = Color3.fromRGB(255, 0, 0)
                                    hitMarker.CFrame = CFrame.new(predictedPos)
                                    hitMarker.Anchored = false
                                    hitMarker.CanCollide = false
                                    hitMarker.Parent = Workspace
                                    local bv = Instance.new("BodyVelocity")
                                    bv.Velocity = Vector3.new(math.random(-5,5), math.random(5,10), math.random(-5,5))
                                    bv.P = 5000
                                    bv.MaxForce = Vector3.new(4000,4000,4000)
                                    bv.Parent = hitMarker
                                    Debris:AddItem(hitMarker, 1.5)
                                else
                                    tracer.Color = Color3.fromRGB(255, 100, 0)
                                end
                            end)
                        end
                    end
                end
            end

            return OriginalFireServer(self, unpack(args))
        end)
    end)
end

-- ─── SPECIAL: Invisibility (ล่องหน) ────────────────────────
local function ActivateInvisibility()
    if not LocalPlayer.Character then return end
    local char = LocalPlayer.Character
    local head = char:FindFirstChild("Head")
    local humanoid = char:FindFirstChild("Humanoid")
    local hrp = char:FindFirstChild("HumanoidRootPart")

    if not head or not humanoid or not hrp then return end

    -- บันทึกตำแหน่งเดิม
    OriginalHeadCFrame = head.CFrame
    OriginalHumanoidCFrame = hrp.CFrame

    -- ย้ายอวตาร (Head + HRP + ทุก part) ลงใต้ดิน
    local offset = Vector3.new(0, InvisibilityOffset, 0)
    for _, part in ipairs(char:GetDescendants()) do
        if part:IsA("BasePart") then
            part.CFrame = part.CFrame + offset
            part.Transparency = 1
            part.CanCollide = false
        end
    end

    -- ซ่อน LocalPlayer character จากผู้เล่นอื่น (Transparency)
    -- มุมกล้องไม่ถูกย้าย — กล้องยังคงอยู่ที่ตำแหน่งเดิมของผู้เล่น
    -- ผู้เล่นยังสามารถโจมตีคนอื่นได้ปกติเพราะ Humanoid ยังทำงาน
end

local function DeactivateInvisibility()
    if not LocalPlayer.Character then return end
    local char = LocalPlayer.Character

    -- คืนตำแหน่งเดิม
    if OriginalHeadCFrame then
        local head = char:FindFirstChild("Head")
        if head then head.CFrame = OriginalHeadCFrame end
    end
    if OriginalHumanoidCFrame then
        local hrp = char:FindFirstChild("HumanoidRootPart")
        if hrp then hrp.CFrame = OriginalHumanoidCFrame end
    end

    -- คืน Transparency
    for _, part in ipairs(char:GetDescendants()) do
        if part:IsA("BasePart") then
            part.Transparency = 0
            part.CanCollide = true
        end
    end

    OriginalHeadCFrame = nil
    OriginalHumanoidCFrame = nil
end

-- ─── SPECIAL: Anti-Lock (กันล็อค) ────────────────────────
local function ActivateAntiLock()
    if not LocalPlayer.Character then return end
    local char = LocalPlayer.Character
    local head = char:FindFirstChild("Head")
    local hrp = char:FindFirstChild("HumanoidRootPart")

    if not head or not hrp then return end

    -- บันทึกตำแหน่งเดิมของหัว
    OriginalHeadCFrame = head.CFrame

    -- ย้ายหัวลงไปในดิน (aimbot ศัตรูจะล็อคที่หัวซึ่งจมดิน)
    head.CFrame = CFrame.new(head.Position.X, hrp.Position.Y - 5, head.Position.Z)
    head.Transparency = 1
    head.CanCollide = false

    -- HumanoidRootPart ยังคงอยู่ที่เดิม — มุมกล้องไม่เปลี่ยน
    -- ผู้เล่นยังโจมตีและยิงได้ปกติ
end

local function DeactivateAntiLock()

    if player == LocalPlayer then return end

-- ─── Auto Attack ───────────────────────────────────────────
local AutoAttackEnabled = false
local AttackCooldown = 0.4

local function GetActiveTool()
    if not LocalPlayer.Character then return nil end
    for _, child in ipairs(LocalPlayer.Character:GetChildren()) do
        if child:IsA("Tool") then return child end
    end
    local backpack = LocalPlayer:FindFirstChild("Backpack")
    if backpack then
        for _, child in ipairs(backpack:GetChildren()) do
            if child:IsA("Tool") then return child end
        end
    end
    return nil
end

local function IsMeleeTool(tool)
    if not tool then return false end
    if tool.Name == "Fists" then return true end
    local meleeFolder = ReplicatedStorage:WaitForChild("Items", 5):WaitForChild("melee", 5)
    local throwableFolder = ReplicatedStorage:WaitForChild("Items", 5):WaitForChild("throwable", 5)
    if meleeFolder:FindFirstChild(tool.Name) and not throwableFolder:FindFirstChild(tool.Name) then
        return true
    end
    return false
end

local function GetPlayersInRange(range)
    local nearby = {}
    if not LocalPlayer.Character or not LocalPlayer.Character.PrimaryPart then return nearby end
    local myPos = LocalPlayer.Character.PrimaryPart.Position
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character and player.Character.PrimaryPart then
            local dist = (player.Character.PrimaryPart.Position - myPos).Magnitude
            if dist <= range then
                table.insert(nearby, player)
            end
        end
    end
    return nearby
end

local function AutoAttackLoop()
    task.spawn(function()
        while AutoAttackEnabled do
            task.wait(AttackCooldown)
            if not LocalPlayer.Character or not LocalPlayer.Character.PrimaryPart then continue end

            local tool = GetActiveTool()
            if not tool or not IsMeleeTool(tool) then continue end

            local nearby = GetPlayersInRange(20)
            if #nearby == 0 then continue end

            local target = nearby[1]
            if target and target.Character and target.Character.PrimaryPart then
                local targetPos = target.Character.PrimaryPart.Position
                local myPos = LocalPlayer.Character.PrimaryPart.Position
                local lookCFrame = CFrame.new(myPos, targetPos)

                if EventCounter and SendRemote then
                    EventCounter.func = (EventCounter.func or 0) + 1
                    pcall(function()
                        SendRemote:FireServer(EventCounter.func, "melee_attack", tool, {target}, lookCFrame, 0.75)
                    end)
                end
            end
        end
    end)
end

-- ─── Melee Aura (Wide Fists) ───────────────────────────────
local OriginalFistAttrs = {}

local function ModifyFists(tool, enable)
    if not tool then return end
    local attrs = tool:GetAttributes()
    local keys = {}
    for k in pairs(attrs) do table.insert(keys, k) end
    table.sort(keys)

    if #keys >= 7 then
        local rangeKey = keys[6]
        local dmgKey = keys[7]
        if enable then
            if OriginalFistAttrs[rangeKey] == nil then
                OriginalFistAttrs[rangeKey] = tool:GetAttribute(rangeKey)
            end
            if OriginalFistAttrs[dmgKey] == nil then
                OriginalFistAttrs[dmgKey] = tool:GetAttribute(dmgKey)
            end
            tool:SetAttribute(rangeKey, 360)
            tool:SetAttribute(dmgKey, 20)
        else
            if OriginalFistAttrs[rangeKey] then tool:SetAttribute(rangeKey, OriginalFistAttrs[rangeKey]) end
            if OriginalFistAttrs[dmgKey] then tool:SetAttribute(dmgKey, OriginalFistAttrs[dmgKey]) end
        end
    end
end

RunService.Heartbeat:Connect(function()
    if not MeleeAura then return end
    if not LocalPlayer.Character then return end
    for _, child in ipairs(LocalPlayer.Character:GetChildren()) do
        if child:IsA("Tool") and IsMeleeTool(child) then
            ModifyFists(child, true)
        end
    end
    local backpack = LocalPlayer:FindFirstChild("Backpack")
    if backpack then
        for _, child in ipairs(backpack:GetChildren()) do
            if child:IsA("Tool") and IsMeleeTool(child) then
                ModifyFists(child, true)
            end
        end
    end
end)

local function TryHoldPrompt(prompt, duration)
    if not prompt or prompt:GetAttribute("__AutoFinishBusy") then return end
    prompt:SetAttribute("__AutoFinishBusy", true)
    pcall(function() prompt:Trigger() end)
    task.wait(duration or 0.05)
    prompt:SetAttribute("__AutoFinishBusy", nil)
end

        if AutoFinish then
            for _, prompt in ipairs(FindFinishPrompts()) do

            end
        end
    end
end)

                    table.sort(servers, function(a, b) return a.playing > b.playing end)
                    game:GetService("TeleportService"):TeleportToPlaceInstance(game.PlaceId, servers[1].id, LocalPlayer)
                end
            end
        end)
    end
})

-- ─── Notification Helper ───────────────────────────────────
local function Notify(title, desc, duration)
    if WindUI and Window and Window.Notify then
        Window:Notify({Title = title, Desc = desc or "", Duration = duration or 3})
    else
        pcall(function()
            game.StarterGui:SetCore("SendNotification", {
                Title = title,
                Text = desc or "",
                Duration = duration or 3
            })
        end)
    end
end
